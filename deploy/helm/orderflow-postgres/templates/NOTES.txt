WARNING: This chart ships with `auth.password=""` (SEC-1 fix). The StatefulSet
will fail its readiness probe until a password is supplied. Provide one via:

  # Option A: inlined in a values override (still secret-stored via the chart-managed Secret)
  helm upgrade --install <release> . \
    --set auth.password=<strong-password>

  # Option B: bring your own Kubernetes Secret
  kubectl create secret generic <my-pg-secret> \
    --from-literal=POSTGRES_USER=orderflow \
    --from-literal=POSTGRES_PASSWORD=<strong-password> \
    --from-literal=POSTGRES_DB=orderflow
  helm upgrade --install <release> . \
    --set secret.create=false \
    --set secret.existingSecret=<my-pg-secret>

MIGRATION FROM v0.2.0 (inlined `auth.password`):
  helm get values <release> -o yaml > /tmp/old-values.yaml
  # Extract the old password:
  OLD_PW=$(yq '.auth.password // "orderflow"' /tmp/old-values.yaml)
  kubectl create secret generic <my-pg-secret> \
    --from-literal=POSTGRES_USER=orderflow \
    --from-literal=POSTGRES_PASSWORD="$OLD_PW" \
    --from-literal=POSTGRES_DB=orderflow \
    --dry-run=client -o yaml | kubectl apply -f -
  helm upgrade <release> . \
    --set secret.create=false \
    --set secret.existingSecret=<my-pg-secret>
  # Then rotate the password on the next maintenance window.
